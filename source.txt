#include "LoginService.h"

LoginService::LoginService(LoginRepository& repo)
    : repository(repo) {
}

QJsonObject LoginService::connecter(
    const QString& login,
    const QString& password
) {
    if (login.trimmed().isEmpty() || password.isEmpty()) {
        return {
            {"success", false},
            {"message", "Login et mot de passe obligatoires"}
        };
    }

    bool trouve = false;

    QString motDePasseEnregistre =
        repository.trouverMotDePasse(login.trimmed(), trouve);

    if (!trouve || motDePasseEnregistre != password) {
        return {
            {"success", false},
            {"message", "Identifiants incorrects"}
        };
    }

    return {
        {"success", true},
        {"message", "Connexion réussie"},
        {"login", login.trimmed()}
    };
}

#include "Initializer.h"

#include <QDir>
#include <QCoreApplication>

bool Initializer::init() {
    QString chemin = "config/application.properties";

    if(!properties.load(chemin)) {
        return false;
    }
    if(!initDatabase()) {
        return false;
    }
    if(!initServeur()) {
        return false;
    }
    if(!initFichierStatiques()) {
        return false;
    }
    if(!initPageRouteur()) {
        return false;
    }   
    if(!initRouteur()) {
        return false;
    }

    routeWeb = std::make_unique<Web>(*pageRouteur);
    routeWeb->declarationDesRoutesWeb();

    routeApi = std::make_unique<Api>(*apiRouteur, base);
    routeApi->declarationDesRoutesApi();

    if (!serveur.lancerServeur()) {
        qCritical() << "Erreur lors du lancement du serveur";
        return false;
    }

return true;


}   

bool Initializer::initDatabase() {
    if(!base.initDatabase(
        properties.get("database.driver"),
        properties.get("database.host"),
        properties.get("database.port").toUShort(),
        properties.get("database.name"),
        properties.get("database.user"),
        properties.get("database.password")   
    )) {
        return false;
    }
    return true;
}

bool Initializer::initServeur() {
    serveur.initServeur(
        properties.get("server.port").toUShort()
    );
    return true;
}

bool Initializer::initFichierStatiques() {
    // Récupérer le dossier Front depuis la configuration
    QString dossier = QDir(
        QCoreApplication::applicationDirPath()
    ).filePath(properties.get("server.static_dir"));

    fichiersStatiques = std::make_unique<StaticFiles>(dossier);

    // Enregistrer le moteur de fichiers sur le serveur HTTP
    fichiersStatiques->enregistrer(serveur.getHttpServer());
    return true;
}

bool Initializer::initPageRouteur() {
    pageRouteur = std::make_unique<PageRouteur>(
        serveur.getHttpServer(),properties.get("server.static_dir")
    );
    return true;
}

bool Initializer::initRouteur() {
    apiRouteur = std::make_unique<Routeur>(serveur.getHttpServer());
    return true;
}

Database& Initializer::getDatabase() {
    return base;
}

Serveur& Initializer::getServeur() {
    return serveur;
}

PageRouteur& Initializer::getPageRouteur() {
    return *pageRouteur;
}

QSqlDatabase Database::getConnection() {
    if(!connection.isOpen()) {
        if(!connection.open()) {
            qCritical() << "Erreur PostgreSQL: " 
                        << connection.lastError().text();
            return QSqlDatabase();
        }
    }
    return connection;
}

bool Database::initDatabase(const QString& driver, const QString& host, const quint16& port, const QString& database,
const QString& user, const QString& password) {
    connection = QSqlDatabase::addDatabase(driver);

    connection.setHostName(host);
    connection.setPort(port);
    connection.setDatabaseName(database);
    connection.setUserName(user);
    connection.setPassword(password);

    return true;
}

void Database::closeConnection() {
    if(connection.isOpen()) {
        connection.close();
    }
}

#include "LoginRepository.h"

#include <QSqlQuery>
#include <QSqlError>
#include <QVariant>
#include <QDebug>

LoginRepository::LoginRepository(Database& database)
    : base(database) {
}

QString LoginRepository::trouverMotDePasse(
    const QString& login,
    bool& trouve
) {
    trouve = false;

    QSqlQuery query(base.getConnection());

    query.prepare(
        "SELECT password FROM Logins WHERE login = :login"
    );

    query.bindValue(":login", login);

    if (!query.exec()) {
        qWarning() << "Erreur SQL :" << query.lastError().text();
        return {};
    }

    if (!query.next()) {
        return {};
    }

    trouve = true;
    return query.value(0).toString();
}

        qCritical() << tcpServer.errorString();

}

#include "StaticFiles.h"

#include <QFile>
#include <QFileInfo>
#include <QMimeDatabase>
#include <QHttpServerResponder>
#include <QDir>

StaticFiles::StaticFiles(const QString& chemin)
    : dossier(QDir(chemin).absolutePath()) {
}

void StaticFiles::enregistrer(QHttpServer& serveur) {
    serveur.setMissingHandler(
        &serveur,
        [this](const QHttpServerRequest& requete, QHttpServerResponder& reponse) {
            QString chemin = requete.url().path();

            if(chemin == "/") {
                chemin = "/index.html";
            }

            QString fichier = QDir(dossier).filePath(chemin.mid(1));
            QFileInfo info(fichier);

            if(!info.exists() || !info.isFile()) {
                reponse.write(
                    "Fichier Introuvable",
                    "text/plain",
                    QHttpServerResponder::StatusCode::NotFound
                );
                return;
            }

            QFile f(fichier);

            if(!f.open(QIODevice::ReadOnly)) {
                reponse.write(
                    "Impossible de lire le fichier",
                    "text/plain",
                    QHttpServerResponder::StatusCode::NotFound
                );
                return;
            }

            QByteArray contenu = f.readAll();

            QMimeDatabase mime;
            QByteArray type = mime.mimeTypeForFile(info).name().toUtf8();

            reponse.write(
                contenu,
                type,
                QHttpServerResponder::StatusCode::Ok
            );
        }
    );
}

#include "PropertiesReader.h"

#include <QFile>
#include <QDebug>
#include <QDir>
#include <QTextStream>

bool PropertiesReader::load(const QString& chemin) {
    QFile fichier(chemin);

    if(!fichier.open(QIODevice::ReadOnly | QIODevice::Text)) {
        qCritical() << "Impossible d'ouvrire le fichier";
        return false;
    }

    QTextStream flux(&fichier);

    while(!flux.atEnd()) {
        QString ligne = flux.readLine().trimmed();
        if(ligne.isEmpty() || ligne.startsWith("#") || ligne.startsWith("!")) {
            continue;
        }
        int separateur = ligne.indexOf("=");
        if(separateur == -1) {
            continue;
        }
        QString clee = ligne.left(separateur).trimmed();
        QString valeur = ligne.mid(separateur+1).trimmed();
        properties.insert(clee,valeur);
    }
    return true;
}

QString PropertiesReader::get(const QString& clee, const QString& valeurParDefaut) const {
    return properties.value(clee,valeurParDefaut);
}

#include "Requete.h"

Requete::Requete(const QHttpServerRequest& req) : requete(req) {}

QString Requete::queryString(const QString& nom, const QString& valeurParDefaut) const {
    QUrlQuery params(requete.url());
    if(!params.hasQueryItem(nom)) {
        return valeurParDefaut;
    }
    return params.queryItemValue(nom);
}

int Requete::queryInt(const QString& nom, const int valeurParDefaut) const {
    bool ok = false;
    QString val = queryString(nom);

    if(val.isEmpty()) {
        return valeurParDefaut;
    }
    int resultat = val.toInt(&ok);
    
    if(!ok) {
        return valeurParDefaut;
    }
    return resultat;
}

bool Requete::queryBool(const QString& nom, const bool valeurParDefaut) const {
    QString valeur = queryString(nom).toLower();
    
    if(valeur == "true" || valeur == "1") {
        return true;
    }
    if(valeur == "false" || valeur == "0") {
        return false;
    }

    return valeurParDefaut;
}

QString Requete::header(const QString& nom) const {
    return QString::fromUtf8(
        requete.value(nom.toUtf8())  
    );
}

QByteArray Requete::body() const {
    return requete.body();
}

QJsonObject Requete::json() const {
    QJsonParseError erreur;
    QJsonDocument document = QJsonDocument::fromJson(requete.body(), &erreur);
    if(erreur.error != QJsonParseError::NoError || !document.isObject()) {
        return {};
    }
    return document.object();
}

#include "PageRouteur.h"

#include <QFile>
#include <QDir>
#include <QHttpServerResponse>

PageRouteur::PageRouteur(QHttpServer& serve, const QString& doc) : serveur(serve), dossier(doc) {}

void PageRouteur::page(const QString& url, const QString& fichier) {
    QString chemin = QDir(dossier).filePath(fichier);

    serveur.route(
        url,
        QHttpServerRequest::Method::Get,
        [chemin]() -> QHttpServerResponse {
            QFile f(chemin);

            if(!f.open(QIODevice::ReadOnly)) {
                return QHttpServerResponse(
                    QHttpServerResponder::StatusCode::NotFound
                );
            }

            return QHttpServerResponse(
                "text/html; charset=utf-8",
                f.readAll()
            );
        }
    );
}
